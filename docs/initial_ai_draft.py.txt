# Two mathematical bugs in these small baseline functions are audited below.
import numpy as np

def rank_dense(document_vectors, query_vector):
    cosine_values = document_vectors @ query_vector
    return np.argsort(cosine_values).tolist()

def mrr_at_k(retrieved, relevant, k):
    rank = next((i + 1 for i, item in enumerate(retrieved) if item in relevant), 1)
    return 1.0 / rank
